### Test Implementation Example

```typescript
// cypress/e2e/membership-claim.cy.ts
describe('Membership Claim Operations', () => {
  beforeEach(() => {
    // Reset blockchain state
    cy.task('resetChopsticks');

    // Clear browser storage
    cy.clearLocalStorage();
    cy.clearCookies();

    // Visit app with local RPC
    cy.visit('/?rpc=ws://localhost:8000');
  });

  describe('Claim Membership', () => {
    it('should allow candidate to claim during claim period', () => {
      // Configure claim period state
      cy.task('setClaimPeriod', {
        round: 6,
        challengeRound: 5,
        candidate: 'Bob',
        approved: true,
      });

      // Connect as approved candidate
      cy.connectWallet('Bob');
      cy.verifyAccountLevel('Candidate');

      // Navigate to journey page with claim flag
      cy.visit('/journey?claim=true&rpc=ws://localhost:8000');

      // Verify claim button is visible
      cy.get('[data-testid="claim-membership-btn"]').should('be.visible');

      // Click Claim Membership
      cy.get('[data-testid="claim-membership-btn"]').click();

      // Confirm transaction
      cy.get('[data-testid="tx-signing"]').should('be.visible');
      cy.get('[data-testid="tx-success"]', { timeout: 30000 }).should('be.visible');

      // Verify success message
      cy.verifyToast('Claim request sent');

      // Verify account level changed to Cyborg
      cy.verifyAccountLevel('Cyborg');
    });

    it('should not allow claim during voting period', () => {
      // Configure voting period state
      cy.task('setVotingPeriod', {
        round: 6,
        challengeRound: 6, // Same round = voting period
        candidate: 'Bob',
      });

      cy.connectWallet('Bob');
      cy.verifyAccountLevel('Candidate');

      cy.visit('/journey?rpc=ws://localhost:8000');

      // Verify claim button not visible
      cy.get('[data-testid="claim-membership-btn"]').should('not.exist');

      // Verify message about waiting
      cy.contains('waiting for voting to end').should('be.visible');

      // Verify period status
      cy.get('[data-testid="voting-period-status"]')
        .should('contain', 'Voting Period');
    });

    it('should not allow non-candidates to claim', () => {
      // Configure claim period
      cy.task('setClaimPeriod', {
        round: 6,
        challengeRound: 5,
        candidate: 'Bob',
        approved: true,
      });

      // Connect as Human
      cy.connectWallet('Dave');
      cy.verifyAccountLevel('Human');

      cy.visit('/journey?rpc=ws://localhost:8000');

      // Verify no claim button
      cy.get('[data-testid="claim-membership-btn"]').should('not.exist');

      // Verify message for non-candidates
      cy.contains('Only candidates can claim').should('be.visible');
    });
  });

  describe('Account Level Transition', () => {
    it('should transition from Candidate to Cyborg after claim', () => {
      // Configure claim period
      cy.task('setClaimPeriod', {
        round: 6,
        challengeRound: 5,
        candidate: 'Bob',
        approved: true,
      });

      cy.connectWallet('Bob');

      // Verify starts as Candidate
      cy.verifyAccountLevel('Candidate');

      // Navigate and claim
      cy.visit('/journey?claim=true&rpc=ws://localhost:8000');
      cy.get('[data-testid="claim-membership-btn"]').click();
      cy.get('[data-testid="tx-success"]', { timeout: 30000 }).should('be.visible');

      // Verify now Cyborg
      cy.verifyAccountLevel('Cyborg');

      // Verify Cyborg capabilities now available
      cy.visitExplore('candidates');

      // Should see vote buttons now
      cy.get('[data-testid^="candidate-approve-btn-"]').should('exist');
      cy.get('[data-testid^="candidate-reject-btn-"]').should('exist');
    });
  });

  describe('Edge Cases', () => {
    it('should handle claim page without wallet connected', () => {
      cy.visit('/journey?claim=true&rpc=ws://localhost:8000');

      // Should see message to connect wallet
      cy.contains('Connect wallet').should('be.visible');

      // No claim button visible
      cy.get('[data-testid="claim-membership-btn"]').should('not.exist');
    });

    it('should not allow rejected candidate to claim', () => {
      // Configure claim period with rejected candidate
      cy.task('setClaimPeriod', {
        round: 6,
        challengeRound: 5,
        candidate: 'Charlie',
        approved: false, // Rejected
      });

      cy.connectWallet('Charlie');
      cy.visit('/journey?rpc=ws://localhost:8000');

      // Verify no claim button
      cy.get('[data-testid="claim-membership-btn"]').should('not.exist');

      // Verify rejection message
      cy.contains('not approved').should('be.visible');
    });

    it('should handle claim after period expires', () => {
      // Configure expired claim period
      cy.task('setExpiredClaimPeriod', {
        round: 7,
        challengeRound: 5,
        candidate: 'Bob',
      });

      cy.connectWallet('Bob');
      cy.visit('/journey?rpc=ws://localhost:8000');

      // Verify no claim button
      cy.get('[data-testid="claim-membership-btn"]').should('not.exist');

      // Verify expiration message
      cy.contains('claim period expired').should('be.visible');
    });

    it('should persist claim status across navigation', () => {
      cy.task('setClaimPeriod', {
        round: 6,
        challengeRound: 5,
        candidate: 'Bob',
        approved: true,
      });

      cy.connectWallet('Bob');
      cy.visit('/journey?claim=true&rpc=ws://localhost:8000');

      // Claim membership
      cy.get('[data-testid="claim-membership-btn"]').click();
      cy.get('[data-testid="tx-success"]', { timeout: 30000 }).should('be.visible');

      // Navigate away
      cy.visitExplore('members');

      // Navigate back
      cy.visit('/journey?rpc=ws://localhost:8000');

      // Verify still Cyborg
      cy.verifyAccountLevel('Cyborg');

      // Claim button should not be visible anymore
      cy.get('[data-testid="claim-membership-btn"]').should('not.exist');
    });

    it('should handle multiple account types attempting to claim', () => {
      cy.task('setClaimPeriod', {
        round: 6,
        challengeRound: 5,
        candidate: 'Bob',
        approved: true,
      });

      const nonCandidateAccounts = [
        { name: 'Dave', level: 'Human' },
        { name: 'Alice', level: 'Bidder' },
        { name: 'Eve', level: 'Cyborg' },
      ];

      nonCandidateAccounts.forEach(({ name, level }) => {
        cy.connectWallet(name);
        cy.verifyAccountLevel(level);

        cy.visit('/journey?rpc=ws://localhost:8000');

        // Verify no claim button for non-candidates
        cy.get('[data-testid="claim-membership-btn"]').should('not.exist');

        // Disconnect for next iteration
        cy.get('[data-testid="connected-account"]').click();
        cy.get('[data-testid="disconnect-btn"]').click();
      });
    });
  });
});
```
