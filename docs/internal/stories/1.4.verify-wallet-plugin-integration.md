### Plugin Configuration Example

```typescript
// cypress/support/e2e.ts
import '@chainsafe/cypress-polkadot-wallet';

// Optional: Configure wallet behavior
Cypress.Commands.overwrite('visit', (originalFn, url, options) => {
  return originalFn(url, {
    ...options,
    onBeforeLoad(win) {
      // Inject wallet extension into window
      // This is handled by the plugin automatically
      if (options?.onBeforeLoad) {
        options.onBeforeLoad(win);
      }
    }
  });
});
```

### Usage Example in Tests

```typescript
// cypress/e2e/wallet-plugin-integration.cy.ts
describe('Wallet Plugin Integration', () => {
  before(() => {
    // Load test accounts
    cy.fixture('accounts').as('accounts');
  });

  it('should inject test accounts into wallet', function() {
    cy.visit('/', {
      onBeforeLoad(win) {
        // Inject Alice account
        cy.injectPolkadotAccount({
          name: this.accounts.alice.name,
          address: this.accounts.alice.address,
          mnemonic: this.accounts.alice.mnemonic
        });
      }
    });

    cy.connectWallet('Alice');
    cy.verifyAccountLevel('Bidder');
  });

  it('should auto-approve transactions', () => {
    cy.visit('/?rpc=ws://localhost:8000');
    cy.connectWallet('Dave');

    // Configure auto-approval
    cy.polkadotWallet().then(wallet => {
      wallet.setAutoApprove(true);
    });

    // Perform transaction
    cy.visitExplore('bidders');
    cy.get('[data-testid="bid-amount-input"]').type('100');
    cy.submitTransaction();

    cy.verifyToast('Bid submitted successfully');
  });
});
```

### Wallet Persistence Pattern

```typescript
it('should persist wallet across navigation', () => {
  cy.visit('/');
  cy.connectWallet('Eve');

  // Navigate to different page
  cy.visit('/explore/members');

  // Wallet should still be connected
  cy.get('[data-testid="connected-account"]').should('be.visible');
  cy.verifyAccountLevel('Cyborg');
});
```
