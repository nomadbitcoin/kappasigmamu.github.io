### Test Implementation Example

```typescript
// cypress/e2e/candidate-voting.cy.ts
describe('Candidate Voting Operations', () => {
  beforeEach(() => {
    // Reset blockchain state
    cy.task('resetChopsticks');

    // Clear browser storage
    cy.clearLocalStorage();
    cy.clearCookies();

    // Visit app with local RPC
    cy.visit('/?rpc=ws://localhost:8000');
  });

  describe('View Candidates', () => {
    it('should display candidate list with details', () => {
      cy.visitExplore('candidates');

      // Verify candidates list is visible
      cy.get('[data-testid="candidates-list"]').should('be.visible');

      // Verify Bob's candidate entry shows all details
      cy.get('[data-testid="candidate-row-Bob"]').within(() => {
        // Address/Identity
        cy.contains('Bob').should('be.visible');

        // Bid type
        cy.contains('Deposit').should('be.visible');

        // Bid amount
        cy.contains('500').should('be.visible');

        // Vote tally
        cy.get('[data-testid="vote-tally-Bob"]')
          .should('contain', 'approvals')
          .and('contain', 'rejections');
      });
    });

    it('should show candidate details panel on click', () => {
      cy.visitExplore('candidates');

      // Click on candidate row
      cy.get('[data-testid="candidate-row-Bob"]').click();

      // Verify detail panel appears
      cy.get('[data-testid="candidate-detail-panel"]').should('be.visible');

      // Verify full information shown
      cy.get('[data-testid="candidate-detail-panel"]').within(() => {
        cy.contains('Bob').should('be.visible');
        cy.contains('Deposit').should('be.visible');
        cy.contains('500').should('be.visible');
      });
    });
  });

  describe('Vote on Candidates', () => {
    it('should allow member to approve a candidate', () => {
      // Connect as member
      cy.connectWallet('Eve');
      cy.verifyAccountLevel('Cyborg');

      cy.visitExplore('candidates');

      // Get initial approval count
      let initialApprovals: number;
      cy.get('[data-testid="vote-tally-Bob"]')
        .invoke('text')
        .then((text) => {
          initialApprovals = parseInt(text.match(/(\d+) approvals?/)?.[1] || '0');
        });

      // Click Approve
      cy.get('[data-testid="candidate-approve-btn-Bob"]').click();

      // Confirm transaction
      cy.get('[data-testid="tx-signing"]').should('be.visible');
      cy.get('[data-testid="tx-success"]', { timeout: 30000 }).should('be.visible');

      // Verify success message
      cy.verifyToast('Approval vote sent');

      // Verify approval count increased
      cy.get('[data-testid="vote-tally-Bob"]')
        .should('contain', `${initialApprovals + 1} approvals`);

      // Verify "Voted" badge appears
      cy.get('[data-testid="candidate-voted-badge-Bob"]').should('be.visible');

      // Verify vote buttons are disabled
      cy.get('[data-testid="candidate-approve-btn-Bob"]').should('be.disabled');
      cy.get('[data-testid="candidate-reject-btn-Bob"]').should('be.disabled');
    });

    it('should allow member to reject a candidate', () => {
      // Connect as member
      cy.connectWallet('Ferdie');
      cy.verifyAccountLevel('Cyborg');

      cy.visitExplore('candidates');

      // Get initial rejection count
      let initialRejections: number;
      cy.get('[data-testid="vote-tally-Charlie"]')
        .invoke('text')
        .then((text) => {
          initialRejections = parseInt(text.match(/(\d+) rejections?/)?.[1] || '0');
        });

      // Click Reject
      cy.get('[data-testid="candidate-reject-btn-Charlie"]').click();

      // Confirm transaction
      cy.get('[data-testid="tx-success"]', { timeout: 30000 }).should('be.visible');

      // Verify success message
      cy.verifyToast('Rejection vote sent');

      // Verify rejection count increased
      cy.get('[data-testid="vote-tally-Charlie"]')
        .should('contain', `${initialRejections + 1} rejections`);

      // Verify "Voted" badge appears
      cy.get('[data-testid="candidate-voted-badge-Charlie"]').should('be.visible');
    });

    it('should not allow non-members to vote', () => {
      // Connect as Human
      cy.connectWallet('Dave');
      cy.verifyAccountLevel('Human');

      cy.visitExplore('candidates');

      // Verify candidates list is visible
      cy.get('[data-testid="candidates-list"]').should('be.visible');

      // Verify no vote buttons visible
      cy.get('[data-testid^="candidate-approve-btn-"]').should('not.exist');
      cy.get('[data-testid^="candidate-reject-btn-"]').should('not.exist');
    });

    it('should prevent member from voting twice', () => {
      cy.connectWallet('Eve');
      cy.visitExplore('candidates');

      // Vote on Bob
      cy.get('[data-testid="candidate-approve-btn-Bob"]').click();
      cy.get('[data-testid="tx-success"]', { timeout: 30000 }).should('be.visible');

      // Verify "Voted" indicator
      cy.get('[data-testid="candidate-voted-badge-Bob"]').should('be.visible');

      // Verify buttons are disabled
      cy.get('[data-testid="candidate-approve-btn-Bob"]').should('be.disabled');
      cy.get('[data-testid="candidate-reject-btn-Bob"]').should('be.disabled');

      // Navigate away and return
      cy.visitExplore('members');
      cy.visitExplore('candidates');

      // Verify vote persists
      cy.get('[data-testid="candidate-voted-badge-Bob"]').should('be.visible');
      cy.get('[data-testid="candidate-approve-btn-Bob"]').should('be.disabled');
    });
  });

  describe('Drop Candidate', () => {
    it('should allow member to drop heavily rejected candidate', () => {
      // Configure Charlie as heavily rejected
      cy.task('setCandidateState', {
        candidate: 'Charlie',
        approvals: 1,
        rejections: 3,
        round: 4, // Current round is 6, so conditions met
      });

      cy.connectWallet('Eve');
      cy.visitExplore('candidates');

      // Verify Drop button is visible
      cy.get('[data-testid="candidate-drop-btn-Charlie"]').should('be.visible');

      // Click Drop
      cy.get('[data-testid="candidate-drop-btn-Charlie"]').click();

      // Confirm transaction
      cy.get('[data-testid="tx-success"]', { timeout: 30000 }).should('be.visible');

      // Verify success message
      cy.verifyToast('Candidate dropped');

      // Verify Charlie removed from list
      cy.get('[data-testid="candidate-row-Charlie"]').should('not.exist');
    });

    it('should not show Drop button when conditions not met', () => {
      cy.visitExplore('candidates');

      // Bob has more approvals than rejections
      cy.get('[data-testid="candidate-row-Bob"]').should('be.visible');

      // Verify no Drop button for Bob
      cy.get('[data-testid="candidate-drop-btn-Bob"]').should('not.exist');
    });
  });

  describe('Edge Cases', () => {
    it('should display candidates without wallet connected', () => {
      cy.visitExplore('candidates');

      // Verify list loads
      cy.get('[data-testid="candidates-list"]').should('be.visible');

      // Verify no vote buttons shown
      cy.get('[data-testid^="candidate-approve-btn-"]').should('not.exist');
      cy.get('[data-testid^="candidate-reject-btn-"]').should('not.exist');
    });

    it('should handle empty candidates list gracefully', () => {
      // Configure Chopsticks with no candidates
      cy.task('clearAllCandidates');

      cy.visitExplore('candidates');

      // Verify appropriate empty state message
      cy.contains('No candidates').should('be.visible');
    });
  });
});
```
