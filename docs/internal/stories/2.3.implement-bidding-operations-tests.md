### Test Implementation Example

```typescript
// cypress/e2e/bidding.cy.ts
describe('Bidding Operations', () => {
  beforeEach(() => {
    // Reset blockchain state
    cy.task('resetChopsticks');

    // Clear browser storage
    cy.clearLocalStorage();
    cy.clearCookies();

    // Visit app with local RPC
    cy.visit('/?rpc=ws://localhost:8000');
  });

  describe('Place a Bid', () => {
    it('should allow Human to place a bid successfully', () => {
      // Connect as Human
      cy.connectWallet('Dave');
      cy.verifyAccountLevel('Human');

      // Navigate to Bidders page
      cy.visitExplore('bidders');

      // Click Bid tab
      cy.get('[data-testid="bid-tab"]').click();

      // Enter bid amount
      cy.get('[data-testid="bid-amount-input"]').type('100');

      // Submit bid
      cy.get('[data-testid="submit-bid-btn"]').click();

      // Verify transaction flow
      cy.get('[data-testid="tx-signing"]').should('be.visible');
      cy.get('[data-testid="tx-success"]', { timeout: 30000 }).should('be.visible');

      // Verify success message
      cy.verifyToast('Bid submitted successfully');

      // Verify bid appears in list
      cy.get('[data-testid="bidders-list"]').should('contain', 'Dave');

      // Verify account level changed
      cy.verifyAccountLevel('Bidder');
    });

    it('should validate minimum bid amount', () => {
      cy.connectWallet('Dave');
      cy.visitExplore('bidders');

      cy.get('[data-testid="bid-tab"]').click();

      // Try to submit with 0 amount
      cy.get('[data-testid="bid-amount-input"]').type('0');
      cy.get('[data-testid="submit-bid-btn"]').click();

      // Verify validation error
      cy.contains('minimum bid').should('be.visible');

      // Verify no transaction submitted
      cy.get('[data-testid="tx-signing"]').should('not.exist');
    });

    it('should validate insufficient balance', () => {
      // Connect with account that has low balance
      cy.connectWallet('LowBalance');
      cy.visitExplore('bidders');

      cy.get('[data-testid="bid-tab"]').click();

      // Enter amount exceeding balance
      cy.get('[data-testid="bid-amount-input"]').type('999999');
      cy.get('[data-testid="submit-bid-btn"]').click();

      // Verify error message
      cy.contains('insufficient', { matchCase: false }).should('be.visible');
    });
  });

  describe('Remove a Bid (Unbid)', () => {
    it('should allow Bidder to remove their bid', () => {
      // Connect as Bidder with active bid
      cy.connectWallet('Alice');
      cy.verifyAccountLevel('Bidder');

      // Navigate to Bidders page
      cy.visitExplore('bidders');

      // Verify bid is highlighted
      cy.get('[data-testid="bid-row-Alice"]').should('have.class', 'highlight');

      // Click Unbid
      cy.get('[data-testid="unbid-btn"]').click();

      // Confirm transaction
      cy.get('[data-testid="tx-success"]', { timeout: 30000 }).should('be.visible');

      // Verify success message
      cy.verifyToast('Bid removed successfully');

      // Verify bid removed from list
      cy.get('[data-testid="bid-row-Alice"]').should('not.exist');

      // Verify account level changed
      cy.verifyAccountLevel('Human');
    });

    it('should not show Unbid button for non-bidders', () => {
      cy.connectWallet('Dave');
      cy.verifyAccountLevel('Human');

      cy.visitExplore('bidders');

      // Verify Unbid button not visible
      cy.get('[data-testid="unbid-btn"]').should('not.exist');
    });
  });

  describe('Vouch for Someone', () => {
    it('should allow Member to vouch for an address', () => {
      // Connect as Member
      cy.connectWallet('Eve');
      cy.verifyAccountLevel('Cyborg');

      // Navigate to Bidders page
      cy.visitExplore('bidders');

      // Click Vouch tab
      cy.get('[data-testid="vouch-tab"]').click();

      // Enter vouch details
      cy.get('[data-testid="vouch-address-input"]').type('5GNJqTPyNqANBkUVMN1LPPrxXnFouWXoe2wNSmmEoLctxiZY');
      cy.get('[data-testid="vouch-amount-input"]').type('50');
      cy.get('[data-testid="vouch-tip-input"]').type('5');

      // Submit vouch
      cy.get('[data-testid="submit-vouch-btn"]').click();

      // Confirm transaction
      cy.get('[data-testid="tx-success"]', { timeout: 30000 }).should('be.visible');

      // Verify success message
      cy.verifyToast('Vouch submitted successfully');

      // Verify vouched address in list
      cy.get('[data-testid="bidders-list"]').should('contain', '5GNJ');

      // Verify shows as Vouch type
      cy.contains('Vouch').should('be.visible');
    });

    it('should allow voucher to remove their vouch', () => {
      // Prerequisite: vouch created (could use cy.task to set state)
      cy.connectWallet('Eve');
      cy.visitExplore('bidders');

      // Find vouch and click Unvouch
      cy.get('[data-testid="unvouch-btn"]').click();

      // Confirm transaction
      cy.get('[data-testid="tx-success"]', { timeout: 30000 }).should('be.visible');

      // Verify success message
      cy.verifyToast('Vouch removed successfully');

      // Verify vouch removed
      cy.get('[data-testid="bidders-list"]').should('not.contain', 'vouched address');
    });
  });

  describe('Account Level Transitions', () => {
    it('should transition from Human to Bidder after placing bid', () => {
      cy.connectWallet('Dave');

      // Verify starts as Human
      cy.verifyAccountLevel('Human');

      // Place bid
      cy.visitExplore('bidders');
      cy.get('[data-testid="bid-tab"]').click();
      cy.get('[data-testid="bid-amount-input"]').type('100');
      cy.get('[data-testid="submit-bid-btn"]').click();
      cy.get('[data-testid="tx-success"]', { timeout: 30000 }).should('be.visible');

      // Verify now Bidder
      cy.verifyAccountLevel('Bidder');
    });

    it('should transition from Bidder to Human after unbid', () => {
      cy.connectWallet('Alice');

      // Verify starts as Bidder
      cy.verifyAccountLevel('Bidder');

      // Unbid
      cy.visitExplore('bidders');
      cy.get('[data-testid="unbid-btn"]').click();
      cy.get('[data-testid="tx-success"]', { timeout: 30000 }).should('be.visible');

      // Verify now Human
      cy.verifyAccountLevel('Human');
    });
  });
});
```
