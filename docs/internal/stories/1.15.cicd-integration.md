### GitHub Actions Workflow Example

```yaml
# .github/workflows/e2e-tests.yml
name: E2E Tests

on:
  pull_request:
    branches: [main, develop]
  push:
    branches: [main]

jobs:
  e2e-tests:
    name: End-to-End Tests
    runs-on: ubuntu-latest
    timeout-minutes: 15

    steps:
      - name: Checkout code
        uses: actions/checkout@v4

      - name: Setup Node.js
        uses: actions/setup-node@v4
        with:
          node-version: '18'
          cache: 'yarn'

      - name: Install dependencies
        run: yarn install --frozen-lockfile

      - name: Cache Cypress binary
        uses: actions/cache@v3
        with:
          path: ~/.cache/Cypress
          key: cypress-${{ runner.os }}-${{ hashFiles('yarn.lock') }}
          restore-keys: |
            cypress-${{ runner.os }}-

      - name: Install Chopsticks
        run: |
          npm install -g @acala-network/chopsticks@latest
          chopsticks --version

      - name: Start Chopsticks
        run: |
          chopsticks --config=config/kusama.yml --port=8000 &
          echo $! > chopsticks.pid
          # Wait for Chopsticks to be ready
          sleep 10
          # Verify Chopsticks is running
          curl -X POST http://localhost:8000 -H "Content-Type: application/json" -d '{"jsonrpc":"2.0","method":"system_health","params":[],"id":1}' || exit 1

      - name: Build application
        run: yarn build
        env:
          CI: true

      - name: Start application server
        run: |
          yarn serve -s build -l 3000 &
          echo $! > app.pid
          # Wait for app to be ready
          npx wait-on http://localhost:3000

      - name: Run Cypress E2E tests
        uses: cypress-io/github-action@v6
        with:
          wait-on: 'http://localhost:3000'
          wait-on-timeout: 60
          browser: chrome
          headless: true
          config: baseUrl=http://localhost:3000
        env:
          CYPRESS_RPC_URL: ws://localhost:8000

      - name: Upload test screenshots
        uses: actions/upload-artifact@v3
        if: failure()
        with:
          name: cypress-screenshots
          path: cypress/screenshots
          retention-days: 7

      - name: Upload test videos
        uses: actions/upload-artifact@v3
        if: always()
        with:
          name: cypress-videos
          path: cypress/videos
          retention-days: 7

      - name: Upload test results
        uses: actions/upload-artifact@v3
        if: always()
        with:
          name: test-results
          path: cypress/results
          retention-days: 7

      - name: Cleanup processes
        if: always()
        run: |
          if [ -f chopsticks.pid ]; then kill $(cat chopsticks.pid) || true; fi
          if [ -f app.pid ]; then kill $(cat app.pid) || true; fi

      - name: Comment PR with results
        uses: actions/github-script@v7
        if: always() && github.event_name == 'pull_request'
        with:
          script: |
            const fs = require('fs');
            const resultsPath = 'cypress/results/results.json';

            if (fs.existsSync(resultsPath)) {
              const results = JSON.parse(fs.readFileSync(resultsPath));
              const comment = `## E2E Test Results

              ✅ Passed: ${results.stats.passes}
              ❌ Failed: ${results.stats.failures}
              ⏭️ Skipped: ${results.stats.pending}
              ⏱️ Duration: ${results.stats.duration}ms
              `;

              github.rest.issues.createComment({
                issue_number: context.issue.number,
                owner: context.repo.owner,
                repo: context.repo.repo,
                body: comment
              });
            }
```

### Alternative: Parallel Execution

```yaml
# For faster execution with parallel tests
jobs:
  e2e-tests:
    name: E2E Tests (Parallel)
    runs-on: ubuntu-latest
    strategy:
      matrix:
        containers: [1, 2, 3]  # Run 3 parallel jobs

    steps:
      # ... (same setup steps)

      - name: Run Cypress tests in parallel
        uses: cypress-io/github-action@v6
        with:
          record: true
          parallel: true
          group: 'E2E Tests'
        env:
          CYPRESS_RECORD_KEY: ${{ secrets.CYPRESS_RECORD_KEY }}
```

### Package.json Scripts

Add these scripts to `package.json`:

```json
{
  "scripts": {
    "chopsticks": "chopsticks --config=config/kusama.yml --port=8000",
    "test:e2e": "cypress open",
    "test:e2e:headless": "cypress run",
    "test:e2e:ci": "concurrently 'yarn chopsticks' 'yarn start' 'wait-on http://localhost:3000 && yarn test:e2e:headless' --kill-others --success first"
  }
}
```

### CI Performance Optimization

1. **Dependency Caching:**
   - Cache `node_modules`
   - Cache Cypress binary
   - Cache build artifacts

2. **Parallel Execution:**
   - Split tests across multiple containers
   - Use Cypress Dashboard for orchestration

3. **Optimized Docker Image (Optional):**
   ```dockerfile
   FROM cypress/browsers:node18.12.0-chrome107
   WORKDIR /app
   COPY package.json yarn.lock ./
   RUN yarn install --frozen-lockfile
   COPY . .
   ```

### Troubleshooting CI Issues

Common issues and solutions:

1. **Chopsticks not starting:**
   - Increase wait time
   - Add health check
   - Check port conflicts

2. **Tests timing out:**
   - Increase `timeout-minutes`
   - Increase `wait-on-timeout`
   - Check network connectivity

3. **Flaky tests:**
   - Add retries: `retries: { runMode: 2 }`
   - Increase timeouts
   - Add explicit waits

4. **Memory issues:**
   - Reduce video quality
   - Disable video recording
   - Use smaller Docker image
