### Test Implementation Example

```typescript
// cypress/e2e/members.cy.ts
describe('Member Operations', () => {
  beforeEach(() => {
    // Reset blockchain state
    cy.task('resetChopsticks');

    // Clear browser storage
    cy.clearLocalStorage();
    cy.clearCookies();

    // Visit app with local RPC
    cy.visit('/?rpc=ws://localhost:8000');
  });

  describe('View Members', () => {
    it('should display member list with details', () => {
      cy.visitExplore('members');

      // Verify members list is visible
      cy.get('[data-testid="members-list"]').should('be.visible');

      // Verify Eve's member entry shows all details
      cy.get('[data-testid="member-row-Eve"]').within(() => {
        // Address/Identity
        cy.contains('Eve').should('be.visible');

        // Account index
        cy.contains('#').should('be.visible');

        // Strike count
        cy.get('[data-testid="member-strikes-Eve"]')
          .should('be.visible')
          .and('contain', '5');

        // Badges
        cy.get('[data-testid="member-badges-Eve"]').should('be.visible');
      });
    });

    it('should highlight member with high strikes as warning', () => {
      cy.visitExplore('members');

      // Verify Eve with 5 strikes has warning styling
      cy.get('[data-testid="member-strikes-Eve"]')
        .should('contain', '5')
        .and('have.class', 'warning'); // or appropriate warning class

      // Could also check CSS color
      cy.get('[data-testid="member-strikes-Eve"]')
        .should('have.css', 'color')
        .and('match', /rgb\(255, 0, 0\)|red/i); // Red color
    });

    it('should show member details panel on click', () => {
      cy.visitExplore('members');

      // Click on member row
      cy.get('[data-testid="member-row-Eve"]').click();

      // Verify detail panel appears
      cy.get('[data-testid="member-detail-panel"]').should('be.visible');

      // Verify full information shown
      cy.get('[data-testid="member-detail-panel"]').within(() => {
        cy.contains('Eve').should('be.visible');
        cy.contains('Account Index').should('be.visible');
        cy.contains('Strikes').should('be.visible');
        cy.contains('5').should('be.visible'); // Strike count
      });
    });
  });

  describe('Defender Voting', () => {
    beforeEach(() => {
      // Configure Chopsticks with active defender
      cy.task('setDefender', { address: 'Eve', round: 6 });
    });

    it('should allow member to approve defender', () => {
      // Connect as member
      cy.connectWallet('Ferdie');
      cy.verifyAccountLevel('Cyborg');

      cy.visitExplore('members');

      // Verify defender section is visible
      cy.get('[data-testid="defender-section"]').should('be.visible');

      // Verify defender member is highlighted
      cy.get('[data-testid="member-row-Eve"]')
        .should('have.class', 'defender'); // or appropriate highlight

      // Click Approve
      cy.get('[data-testid="defender-approve-btn"]').click();

      // Confirm transaction
      cy.get('[data-testid="tx-signing"]').should('be.visible');
      cy.get('[data-testid="tx-success"]', { timeout: 30000 }).should('be.visible');

      // Verify success message
      cy.verifyToast('Approval vote sent');

      // Verify vote recorded
      cy.get('[data-testid="defender-section"]')
        .should('contain', 'Your vote: Approve');
    });

    it('should allow member to reject defender', () => {
      // Connect as member
      cy.connectWallet('Eve');
      cy.verifyAccountLevel('Cyborg');

      cy.visitExplore('members');

      // Verify defender section is visible
      cy.get('[data-testid="defender-section"]').should('be.visible');

      // Click Reject
      cy.get('[data-testid="defender-reject-btn"]').click();

      // Confirm transaction
      cy.get('[data-testid="tx-success"]', { timeout: 30000 }).should('be.visible');

      // Verify success message
      cy.verifyToast('Rejection vote sent');

      // Verify vote recorded
      cy.get('[data-testid="defender-section"]')
        .should('contain', 'Your vote: Reject');
    });

    it('should not allow non-members to vote on defender', () => {
      // Connect as Human
      cy.connectWallet('Dave');
      cy.verifyAccountLevel('Human');

      cy.visitExplore('members');

      // Verify defender section is visible (can see defender)
      cy.get('[data-testid="defender-section"]').should('be.visible');

      // Verify member is highlighted as defender
      cy.get('[data-testid="member-row-Eve"]')
        .should('have.class', 'defender');

      // Verify vote buttons not visible
      cy.get('[data-testid="defender-approve-btn"]').should('not.exist');
      cy.get('[data-testid="defender-reject-btn"]').should('not.exist');

      // Verify message about member-only feature
      cy.contains('Members only').should('be.visible');
    });
  });

  describe('Edge Cases', () => {
    it('should display members without wallet connected', () => {
      cy.visitExplore('members');

      // Verify list loads
      cy.get('[data-testid="members-list"]').should('be.visible');

      // Verify members displayed
      cy.get('[data-testid="member-row-Eve"]').should('be.visible');
      cy.get('[data-testid="member-row-Ferdie"]').should('be.visible');
    });

    it('should display various member badges correctly', () => {
      cy.visitExplore('members');

      // Check Eve's badges (Head)
      cy.get('[data-testid="member-badges-Eve"]')
        .should('contain', 'Head');

      // Check for other badge types if configured
      // Founder, Defender, etc.
    });

    it('should differentiate between low and high strikes', () => {
      cy.visitExplore('members');

      // Ferdie has 0 strikes (no warning)
      cy.get('[data-testid="member-strikes-Ferdie"]')
        .should('contain', '0')
        .and('not.have.class', 'warning');

      // Eve has 5 strikes (warning)
      cy.get('[data-testid="member-strikes-Eve"]')
        .should('contain', '5')
        .and('have.class', 'warning');
    });

    it('should handle no active defender gracefully', () => {
      // Configure no defender
      cy.task('clearDefender');

      cy.connectWallet('Ferdie');
      cy.visitExplore('members');

      // Verify defender section shows no defender message
      cy.get('[data-testid="defender-section"]')
        .should('contain', 'No defender');

      // Or defender section not visible
      // cy.get('[data-testid="defender-section"]').should('not.exist');
    });

    it('should display member rank information', () => {
      cy.visitExplore('members');

      // Check Ferdie's rank
      cy.get('[data-testid="member-row-Ferdie"]')
        .should('contain', 'Rank')
        .and('contain', '1');

      // Check Eve's rank
      cy.get('[data-testid="member-row-Eve"]')
        .should('contain', 'Rank')
        .and('contain', '0');
    });
  });
});
```
