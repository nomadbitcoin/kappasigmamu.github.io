### Test Implementation Example

```typescript
// cypress/e2e/error-handling.cy.ts
describe('Error Handling & Edge Cases', () => {
  beforeEach(() => {
    // Reset blockchain state
    cy.task('resetChopsticks');

    // Clear browser storage
    cy.clearLocalStorage();
    cy.clearCookies();

    // Visit app with local RPC
    cy.visit('/?rpc=ws://localhost:8000');
  });

  describe('Transaction Failures', () => {
    it('should handle transaction rejected by user', () => {
      cy.connectWallet('Dave');
      cy.visitExplore('bidders');

      // Start bid process
      cy.get('[data-testid="bid-tab"]').click();
      cy.get('[data-testid="bid-amount-input"]').type('100');
      cy.get('[data-testid="submit-bid-btn"]').click();

      // Simulate wallet rejection
      cy.task('rejectTransaction');

      // Verify error message
      cy.get('[data-testid="tx-error"]').should('be.visible');
      cy.verifyToast('Transaction cancelled');

      // Verify no changes made
      cy.verifyAccountLevel('Human');
      cy.get('[data-testid="bidders-list"]').should('not.contain', 'Dave');

      // Verify can retry
      cy.get('[data-testid="submit-bid-btn"]').should('be.enabled');
    });

    it('should handle transaction failure on chain', () => {
      // Connect with low-balance account
      cy.task('setAccountBalance', { account: 'Dave', balance: 1 });
      cy.connectWallet('Dave');
      cy.visitExplore('bidders');

      // Try to place bid exceeding balance
      cy.get('[data-testid="bid-tab"]').click();
      cy.get('[data-testid="bid-amount-input"]').type('100');
      cy.get('[data-testid="submit-bid-btn"]').click();

      // Transaction should fail
      cy.get('[data-testid="tx-error"]', { timeout: 30000 }).should('be.visible');

      // Verify error message
      cy.contains('insufficient', { matchCase: false }).should('be.visible');

      // Verify state unchanged
      cy.verifyAccountLevel('Human');
    });

    it('should handle network disconnection during transaction', () => {
      cy.connectWallet('Dave');
      cy.visitExplore('bidders');

      // Intercept RPC calls to simulate disconnection
      cy.intercept('POST', '**/ws', { forceNetworkError: true }).as('networkError');

      // Try to place bid
      cy.get('[data-testid="bid-tab"]').click();
      cy.get('[data-testid="bid-amount-input"]').type('100');
      cy.get('[data-testid="submit-bid-btn"]').click();

      // Verify error message about connection
      cy.contains('connection', { matchCase: false }).should('be.visible');

      // Verify prompted to reconnect
      cy.contains('reconnect', { matchCase: false }).should('be.visible');
    });
  });

  describe('UI Edge Cases', () => {
    it('should load pages without wallet connected', () => {
      const sections = ['bidders', 'candidates', 'members', 'payouts'];

      sections.forEach((section) => {
        cy.visitExplore(section);

        // Verify page loads (read-only)
        cy.get(`[data-testid="${section}-list"]`).should('be.visible');

        // Verify no action buttons
        cy.get('[data-testid*="btn"]').should('not.exist');

        // Verify prompt to connect
        cy.contains('Connect wallet').should('be.visible');
      });
    });

    it('should handle refresh during transaction', () => {
      cy.connectWallet('Dave');
      cy.visitExplore('bidders');

      // Start transaction
      cy.get('[data-testid="bid-tab"]').click();
      cy.get('[data-testid="bid-amount-input"]').type('100');
      cy.get('[data-testid="submit-bid-btn"]').click();

      // Wait for signing prompt
      cy.get('[data-testid="tx-signing"]').should('be.visible');

      // Refresh page
      cy.reload();

      // Verify need to reconnect
      cy.get('[data-testid="connect-wallet-btn"]').should('be.visible');

      // Reconnect
      cy.connectWallet('Dave');

      // Should be able to check status
      cy.visitExplore('bidders');
      cy.waitForBlockchainData();

      // Can retry if needed
      cy.get('[data-testid="bid-tab"]').click();
    });

    it('should validate form inputs', () => {
      cy.connectWallet('Dave');
      cy.visitExplore('bidders');

      cy.get('[data-testid="bid-tab"]').click();

      // Test empty input
      cy.get('[data-testid="submit-bid-btn"]').click();
      cy.contains('required', { matchCase: false }).should('be.visible');

      // Test invalid input
      cy.get('[data-testid="bid-amount-input"]').type('abc');
      cy.get('[data-testid="submit-bid-btn"]').click();
      cy.contains('valid number').should('be.visible');

      // Test negative number
      cy.get('[data-testid="bid-amount-input"]').clear().type('-10');
      cy.get('[data-testid="submit-bid-btn"]').click();
      cy.contains('positive').should('be.visible');

      // Test zero
      cy.get('[data-testid="bid-amount-input"]').clear().type('0');
      cy.get('[data-testid="submit-bid-btn"]').click();
      cy.contains('minimum').should('be.visible');
    });

    it('should handle rapid button clicking', () => {
      cy.connectWallet('Dave');
      cy.visitExplore('bidders');

      cy.get('[data-testid="bid-tab"]').click();
      cy.get('[data-testid="bid-amount-input"]').type('100');

      // Click submit multiple times rapidly
      cy.get('[data-testid="submit-bid-btn"]').click();
      cy.get('[data-testid="submit-bid-btn"]').click();
      cy.get('[data-testid="submit-bid-btn"]').click();

      // Should only submit once
      cy.get('[data-testid="tx-signing"]').should('be.visible');

      // Button should be disabled during transaction
      cy.get('[data-testid="submit-bid-btn"]').should('be.disabled');
    });

    it('should handle navigation during loading', () => {
      cy.visitExplore('bidders');

      // Immediately navigate away while loading
      cy.visit('/explore/candidates?rpc=ws://localhost:8000');

      // Should cancel previous load and load new page
      cy.url().should('include', '/candidates');
      cy.waitForBlockchainData();
      cy.get('[data-testid="candidates-list"]').should('be.visible');
    });
  });

  describe('Blockchain Error Scenarios', () => {
    it('should handle insufficient funds error', () => {
      cy.task('setAccountBalance', { account: 'Dave', balance: 10 });
      cy.connectWallet('Dave');
      cy.visitExplore('bidders');

      cy.get('[data-testid="bid-tab"]').click();
      cy.get('[data-testid="bid-amount-input"]').type('100');
      cy.get('[data-testid="submit-bid-btn"]').click();

      // Verify specific error message
      cy.get('[data-testid="tx-error"]', { timeout: 30000 }).should('be.visible');
      cy.contains('insufficient funds', { matchCase: false }).should('be.visible');
    });

    it('should handle invalid parameters error', () => {
      cy.connectWallet('Dave');
      cy.visitExplore('bidders');

      // Simulate invalid parameter via direct API call
      cy.task('submitInvalidBid', { account: 'Dave', amount: -100 });

      // Verify error handling
      cy.contains('invalid', { matchCase: false }).should('be.visible');
    });

    it('should handle permission denied errors', () => {
      // Try member-only action as non-member
      cy.connectWallet('Dave'); // Human
      cy.visitExplore('candidates');

      // Vote buttons should not be visible (permission prevention)
      cy.get('[data-testid^="candidate-approve-btn-"]').should('not.exist');
      cy.get('[data-testid^="candidate-reject-btn-"]').should('not.exist');
    });
  });
});
```

- Tests must be deterministic (no flaky tests)

- Must verify UI state after errors
- Must test error recovery and retry scenarios
