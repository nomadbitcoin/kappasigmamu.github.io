### Test Implementation Example

```typescript
// cypress/e2e/payouts.cy.ts
describe('Payout Operations', () => {
  beforeEach(() => {
    // Reset blockchain state
    cy.task('resetChopsticks');

    // Clear browser storage
    cy.clearLocalStorage();
    cy.clearCookies();

    // Visit app with local RPC
    cy.visit('/?rpc=ws://localhost:8000');
  });

  describe('View Payouts', () => {
    it('should display payout list with details', () => {
      cy.visitExplore('payouts');

      // Verify payouts list is visible
      cy.get('[data-testid="payouts-list"]').should('be.visible');

      // Verify payout entry shows all details
      cy.get('[data-testid="payout-row-Ferdie"]').within(() => {
        // Address/Identity
        cy.contains('Ferdie').should('be.visible');

        // Total paid amount
        cy.get('[data-testid="payout-total-Ferdie"]').should('be.visible');

        // Pending amount
        cy.get('[data-testid="payout-pending-Ferdie"]').should('be.visible');

        // Maturity status
        cy.get('[data-testid="payout-maturity-Ferdie"]').should('be.visible');
      });
    });

    it('should show maturity countdown for unmatured payout', () => {
      // Configure Chopsticks with unmatured payout
      cy.task('setPayoutMaturity', { account: 'Ferdie', blocksRemaining: 100 });

      cy.visitExplore('payouts');

      // Verify countdown is displayed
      cy.get('[data-testid="payout-maturity-Ferdie"]')
        .should('contain', 'Maturing in')
        .and('contain', 'blocks');

      // Verify no claim button
      cy.get('[data-testid="claim-payout-btn-Ferdie"]').should('not.exist');
    });

    it('should show matured badge and claim button for matured payout', () => {
      // Configure Chopsticks with matured payout
      cy.task('setPayoutMaturity', { account: 'Ferdie', blocksRemaining: 0 });

      cy.visitExplore('payouts');

      // Verify "Matured" badge
      cy.get('[data-testid="payout-maturity-Ferdie"]')
        .should('contain', 'Matured');

      // Note: Claim button only visible when connected as the member
    });
  });

  describe('Claim Payout', () => {
    it('should allow member to claim matured payout', () => {
      // Configure matured payout
      cy.task('setPayoutMaturity', { account: 'Ferdie', blocksRemaining: 0 });

      // Connect as Ferdie
      cy.connectWallet('Ferdie');
      cy.verifyAccountLevel('Cyborg');

      // Get initial balance
      let initialBalance: string;
      cy.get('[data-testid="account-balance"]').invoke('text').then((balance) => {
        initialBalance = balance;
      });

      // Navigate to payouts
      cy.visitExplore('payouts');

      // Verify claim button is visible
      cy.get('[data-testid="claim-payout-btn-Ferdie"]').should('be.visible');

      // Click claim
      cy.get('[data-testid="claim-payout-btn-Ferdie"]').click();

      // Confirm transaction
      cy.get('[data-testid="tx-signing"]').should('be.visible');
      cy.get('[data-testid="tx-success"]', { timeout: 30000 }).should('be.visible');

      // Verify success message
      cy.verifyToast('Payout claimed successfully');

      // Verify payout status updated
      cy.get('[data-testid="payout-pending-Ferdie"]')
        .should('not.contain', '15,000'); // Original pending amount

      // Verify balance increased
      cy.get('[data-testid="account-balance"]').invoke('text').should((newBalance) => {
        expect(newBalance).not.to.equal(initialBalance);
      });
    });

    it('should not allow claiming unmatured payout', () => {
      // Configure unmatured payout
      cy.task('setPayoutMaturity', { account: 'Ferdie', blocksRemaining: 50 });

      cy.connectWallet('Ferdie');
      cy.visitExplore('payouts');

      // Verify countdown is shown
      cy.get('[data-testid="payout-maturity-Ferdie"]')
        .should('contain', 'Maturing in');

      // Verify claim button not available
      cy.get('[data-testid="claim-payout-btn-Ferdie"]').should('not.exist');
    });

    it('should not show claim buttons for non-members', () => {
      // Configure matured payout
      cy.task('setPayoutMaturity', { account: 'Ferdie', blocksRemaining: 0 });

      // Connect as Human (not a member)
      cy.connectWallet('Dave');
      cy.verifyAccountLevel('Human');

      cy.visitExplore('payouts');

      // Verify payouts list is visible (read-only)
      cy.get('[data-testid="payouts-list"]').should('be.visible');

      // Verify no claim buttons visible
      cy.get('[data-testid^="claim-payout-btn-"]').should('not.exist');
    });
  });

  describe('Edge Cases', () => {
    it('should display payouts without wallet connected', () => {
      cy.visitExplore('payouts');

      // Verify list loads
      cy.get('[data-testid="payouts-list"]').should('be.visible');

      // Verify no claim buttons shown
      cy.get('[data-testid^="claim-payout-btn-"]').should('not.exist');
    });

    it('should handle empty payout list gracefully', () => {
      // Configure Chopsticks with no pending payouts
      cy.task('clearAllPayouts');

      cy.visitExplore('payouts');

      // Verify appropriate empty state message
      cy.contains('No pending payouts').should('be.visible');
    });

    it('should display multiple payouts for same member', () => {
      // Configure multiple payouts for Ferdie
      cy.task('setMultiplePayouts', {
        account: 'Ferdie',
        payouts: [
          { amount: 10000, matured: true },
          { amount: 5000, matured: false }
        ]
      });

      cy.connectWallet('Ferdie');
      cy.visitExplore('payouts');

      // Verify both payouts shown
      cy.get('[data-testid="payout-row-Ferdie"]').should('be.visible');
      cy.get('[data-testid="payout-pending-Ferdie"]')
        .should('contain', '15,000'); // Total of both
    });
  });
});
```
