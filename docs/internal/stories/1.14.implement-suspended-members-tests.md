### Test Implementation Example

```typescript
// cypress/e2e/suspended.cy.ts
describe('Suspended Members', () => {
  beforeEach(() => {
    // Reset blockchain state
    cy.task('resetChopsticks');

    // Clear browser storage
    cy.clearLocalStorage();
    cy.clearCookies();

    // Visit app with local RPC
    cy.visit('/?rpc=ws://localhost:8000');
  });

  describe('View Suspended Members', () => {
    it('should display suspended members list', () => {
      // Configure Chopsticks with suspended members
      cy.task('setSuspendedMembers', [
        {
          address: 'Alice',
          reason: 'Exceeded maximum strikes',
          block: 1000000,
        },
        {
          address: 'Charlie',
          reason: 'Failed to defend',
          block: 1001000,
        },
      ]);

      cy.visitExplore('suspended');

      // Verify list is visible
      cy.get('[data-testid="suspended-list"]').should('be.visible');

      // Verify suspended members shown
      cy.get('[data-testid="suspended-row-Alice"]').should('be.visible');
      cy.get('[data-testid="suspended-row-Charlie"]').should('be.visible');

      // Verify details shown
      cy.get('[data-testid="suspended-row-Alice"]').within(() => {
        cy.contains('Alice').should('be.visible');
        cy.contains('Exceeded maximum strikes').should('be.visible');
      });
    });

    it('should show single suspended member', () => {
      cy.task('setSuspendedMembers', [
        {
          address: 'Alice',
          reason: 'Exceeded maximum strikes',
        },
      ]);

      cy.visitExplore('suspended');

      cy.get('[data-testid="suspended-list"]').should('be.visible');
      cy.get('[data-testid="suspended-row-Alice"]').should('be.visible');

      // Should not show empty state
      cy.get('[data-testid="empty-state-message"]').should('not.exist');
    });
  });

  describe('Empty Suspended List', () => {
    it('should show appropriate message when no suspended members', () => {
      // Configure Chopsticks with no suspended members
      cy.task('setSuspendedMembers', []);

      cy.visitExplore('suspended');

      // Verify empty state message
      cy.get('[data-testid="empty-state-message"]').should('be.visible');

      // Verify appropriate messaging
      cy.contains(/no suspended members|all members are in good standing/i)
        .should('be.visible');

      // Should not show list
      cy.get('[data-testid="suspended-list"]').should('not.exist');

      // Should not show error
      cy.get('[data-testid="tx-error"]').should('not.exist');
    });
  });

  describe('Suspended Members Details', () => {
    it('should show suspended member details if panel exists', () => {
      cy.task('setSuspendedMembers', [
        {
          address: 'Alice',
          reason: 'Exceeded maximum strikes',
          strikes: 6,
          block: 1000000,
        },
      ]);

      cy.visitExplore('suspended');

      // Click on suspended member (if clickable)
      cy.get('[data-testid="suspended-row-Alice"]').click();

      // Verify detail panel (if exists)
      cy.get('[data-testid="member-detail-panel"]').should('be.visible');

      // Verify full information
      cy.get('[data-testid="member-detail-panel"]').within(() => {
        cy.contains('Alice').should('be.visible');
        cy.contains('6').should('be.visible'); // Strikes
        cy.contains('Exceeded maximum strikes').should('be.visible');
      });
    });
  });

  describe('Edge Cases', () => {
    it('should display suspended page without wallet connected', () => {
      cy.task('setSuspendedMembers', [
        { address: 'Alice', reason: 'Test' },
      ]);

      cy.visitExplore('suspended');

      // Verify page loads
      cy.get('[data-testid="suspended-list"]').should('be.visible');

      // No wallet needed for viewing
      cy.get('[data-testid="connect-wallet-btn"]').should('be.visible');
    });

    it('should handle navigation to and from suspended page', () => {
      cy.visitExplore('members');
      cy.get('[data-testid="members-list"]').should('be.visible');

      // Navigate to suspended
      cy.get('[data-testid="nav-link-suspended"]').click();
      cy.url().should('include', '/suspended');

      // Navigate back
      cy.get('[data-testid="nav-link-members"]').click();
      cy.url().should('include', '/members');
    });

    it('should handle suspended member with missing identity', () => {
      cy.task('setSuspendedMembers', [
        {
          address: '5GrwvaEF5zXb26Fz9rcQpDWS57CtERHpNehXCPcNoHGKutQY',
          reason: 'Test',
          // No identity set
        },
      ]);

      cy.visitExplore('suspended');

      // Should show truncated address
      cy.contains('5Grw...utQY').should('be.visible');
    });

    it('should handle very long suspension reason', () => {
      const longReason = 'This is a very long suspension reason that might need to be truncated or wrapped properly in the UI to ensure good user experience and readability';

      cy.task('setSuspendedMembers', [
        {
          address: 'Alice',
          reason: longReason,
        },
      ]);

      cy.visitExplore('suspended');

      // Verify reason is displayed
      cy.contains(longReason).should('be.visible');

      // Could also check if it's properly truncated or wrapped
      // cy.get('[data-testid="suspension-reason"]').should('have.css', 'overflow', 'hidden');
    });

    it('should handle multiple suspended members', () => {
      const suspendedMembers = [
        { address: 'Alice', reason: 'Reason 1' },
        { address: 'Bob', reason: 'Reason 2' },
        { address: 'Charlie', reason: 'Reason 3' },
        { address: 'Dave', reason: 'Reason 4' },
      ];

      cy.task('setSuspendedMembers', suspendedMembers);

      cy.visitExplore('suspended');

      // Verify all members shown
      suspendedMembers.forEach(({ address }) => {
        cy.get(`[data-testid="suspended-row-${address}"]`).should('be.visible');
      });
    });

    it('should load suspended page with slow network', () => {
      // Throttle network
      cy.intercept('POST', '**/ws', (req) => {
        req.on('response', (res) => {
          res.setDelay(2000); // 2 second delay
        });
      });

      cy.visitExplore('suspended');

      // Should show loading state
      cy.get('[data-testid="loading-spinner"]').should('be.visible');

      // Eventually loads
      cy.get('[data-testid="blockchain-data"]', { timeout: 10000 }).should('be.visible');
    });
  });
});
```
