### Implementation Examples

#### connectWallet Command
```typescript
Cypress.Commands.add('connectWallet', (accountName: string) => {
  cy.get('[data-testid="connect-wallet-btn"]').click();
  cy.get('[data-testid="wallet-polkadot"]').click();
  cy.get(`[data-testid="account-${accountName}"]`).click();
  cy.get('[data-testid="connected-account"]').should('be.visible');
});
```

#### TypeScript Declaration
```typescript
// cypress/support/types.d.ts
declare namespace Cypress {
  interface Chainable {
    /**
     * Connect wallet with specified account
     * @param accountName - Name of test account (Alice, Bob, etc.)
     * @example cy.connectWallet('Alice')
     */
    connectWallet(accountName: string): Chainable<void>;

    /**
     * Wait for blockchain data to finish loading
     * @param timeout - Optional timeout in ms (default 10000)
     * @example cy.waitForBlockchainData()
     */
    waitForBlockchainData(timeout?: number): Chainable<void>;

    // ... other commands
  }
}
```
