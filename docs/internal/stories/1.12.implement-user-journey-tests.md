### Test Implementation Example

```typescript
// cypress/e2e/user-journeys.cy.ts
describe('User Journey Flows', () => {
  beforeEach(() => {
    // Reset blockchain state
    cy.task('resetChopsticks');

    // Clear browser storage
    cy.clearLocalStorage();
    cy.clearCookies();

    // Visit app with local RPC
    cy.visit('/?rpc=ws://localhost:8000');
  });

  describe('New User Journey: Human → Bidder', () => {
    it('should complete full journey from visitor to bidder', () => {
      // Step 1: Start as visitor
      cy.url().should('include', '/');

      // Step 2: Explore the site
      cy.visit('/welcome?rpc=ws://localhost:8000');
      cy.contains('Welcome').should('be.visible');

      cy.visit('/journey?rpc=ws://localhost:8000');
      cy.contains('Journey').should('be.visible');

      // Step 3: Connect wallet
      cy.get('[data-testid="connect-wallet-btn"]').click();
      cy.get('[data-testid="wallet-modal"]').should('be.visible');
      cy.get('[data-testid="wallet-polkadot"]').click();
      cy.get('[data-testid="account-Dave"]').click();

      // Step 4: Verify Human status
      cy.verifyAccountLevel('Human');
      cy.contains('submit a bid').should('be.visible');

      // Step 5: Navigate to Bidders page
      cy.visitExplore('bidders');

      // Step 6: Place bid
      cy.get('[data-testid="bid-tab"]').click();
      cy.get('[data-testid="bid-amount-input"]').type('100');
      cy.get('[data-testid="submit-bid-btn"]').click();

      // Step 7: Approve transaction
      cy.get('[data-testid="tx-signing"]').should('be.visible');
      cy.get('[data-testid="tx-success"]', { timeout: 30000 }).should('be.visible');

      // Step 8: Verify Bidder status
      cy.verifyAccountLevel('Bidder');
      cy.verifyToast('Bid submitted successfully');

      // Step 9: Verify bid in list
      cy.get('[data-testid="bidders-list"]').should('contain', 'Dave');

      // Step 10: Verify unbid option available
      cy.get('[data-testid="unbid-btn"]').should('be.visible');
    });
  });

  describe('Member Participation Journey', () => {
    it('should complete full governance participation flow', () => {
      // Step 1: Connect as member
      cy.connectWallet('Eve');
      cy.verifyAccountLevel('Cyborg');

      // Step 2: Navigate to Candidates page
      cy.visitExplore('candidates');

      // Step 3: Vote to approve candidate
      cy.get('[data-testid="candidate-approve-btn-Bob"]').click();
      cy.get('[data-testid="tx-success"]', { timeout: 30000 }).should('be.visible');
      cy.verifyToast('Approval vote sent');

      // Step 4: Verify vote recorded
      cy.get('[data-testid="candidate-voted-badge-Bob"]').should('be.visible');

      // Step 5: Navigate to Members page
      cy.visitExplore('members');

      // Step 6: Vote on defender (if active)
      cy.task('setDefender', { address: 'Ferdie', round: 6 });
      cy.reload();
      cy.waitForBlockchainData();

      cy.get('[data-testid="defender-approve-btn"]').click();
      cy.get('[data-testid="tx-success"]', { timeout: 30000 }).should('be.visible');

      // Step 7: Navigate to Payouts page
      cy.visitExplore('payouts');

      // Step 8: View payout information
      cy.get('[data-testid="payouts-list"]').should('be.visible');
      cy.get('[data-testid="payout-row-Eve"]').should('be.visible');

      // Verify complete governance participation
      cy.log('Member successfully participated in all governance features');
    });
  });

  describe('Bidder Lifecycle: Bid → Unbid → Rebid', () => {
    it('should complete full bid lifecycle', () => {
      // Step 1: Connect as existing bidder
      cy.connectWallet('Alice');
      cy.verifyAccountLevel('Bidder');

      // Step 2: Navigate to Bidders page
      cy.visitExplore('bidders');

      // Step 3: Verify active bid
      cy.get('[data-testid="bid-row-Alice"]').should('be.visible');
      cy.get('[data-testid="unbid-btn"]').should('be.visible');

      // Step 4: Unbid
      cy.get('[data-testid="unbid-btn"]').click();
      cy.get('[data-testid="tx-success"]', { timeout: 30000 }).should('be.visible');
      cy.verifyToast('Bid removed successfully');

      // Step 5: Verify level changed to Human
      cy.verifyAccountLevel('Human');

      // Step 6: Verify bid removed
      cy.get('[data-testid="bid-row-Alice"]').should('not.exist');

      // Step 7: Place new bid with different amount
      cy.get('[data-testid="bid-tab"]').click();
      cy.get('[data-testid="bid-amount-input"]').type('150');
      cy.get('[data-testid="submit-bid-btn"]').click();

      // Step 8: Approve transaction
      cy.get('[data-testid="tx-success"]', { timeout: 30000 }).should('be.visible');
      cy.verifyToast('Bid submitted successfully');

      // Step 9: Verify level changed back to Bidder
      cy.verifyAccountLevel('Bidder');

      // Step 10: Verify new bid amount
      cy.get('[data-testid="bid-row-Alice"]').should('contain', '150');
    });
  });

  describe('Journey Edge Cases', () => {
    it('should handle wallet disconnect mid-journey', () => {
      cy.connectWallet('Dave');
      cy.verifyAccountLevel('Human');

      cy.visitExplore('bidders');

      // Disconnect wallet
      cy.get('[data-testid="connected-account"]').click();
      cy.get('[data-testid="disconnect-btn"]').click();

      // Verify UI adapts
      cy.get('[data-testid="submit-bid-btn"]').should('not.exist');
      cy.get('[data-testid="connect-wallet-btn"]').should('be.visible');
    });

    it('should maintain state across page refresh', () => {
      cy.connectWallet('Eve');
      cy.verifyAccountLevel('Cyborg');

      // Navigate to candidates and vote
      cy.visitExplore('candidates');
      cy.get('[data-testid="candidate-approve-btn-Bob"]').click();
      cy.get('[data-testid="tx-success"]', { timeout: 30000 }).should('be.visible');

      // Refresh page
      cy.reload();

      // Verify wallet still connected
      cy.verifyAccountLevel('Cyborg');

      // Verify vote persisted
      cy.get('[data-testid="candidate-voted-badge-Bob"]').should('be.visible');
    });

    it('should handle navigation between all major sections', () => {
      cy.connectWallet('Ferdie');

      const sections = ['bidders', 'candidates', 'members', 'payouts'];

      sections.forEach((section) => {
        cy.visitExplore(section);
        cy.url().should('include', `/explore/${section}`);
        cy.waitForBlockchainData();

        // Verify wallet still connected
        cy.get('[data-testid="connected-account"]').should('be.visible');
      });
    });
  });
});
```
