### Test Implementation Example

```typescript
// cypress/e2e/navigation.cy.ts
describe('Navigation & Page Access', () => {
  beforeEach(() => {
    // Reset blockchain state
    cy.task('resetChopsticks');

    // Clear browser storage
    cy.clearLocalStorage();
    cy.clearCookies();

    // Visit app with local RPC
    cy.visit('/?rpc=ws://localhost:8000');
  });

  describe('Primary Routes', () => {
    const primaryRoutes = [
      { route: '/', page: 'Landing', content: 'Kusama Society' },
      { route: '/welcome', page: 'Welcome', content: 'Welcome' },
      { route: '/journey', page: 'Journey', content: 'Journey' },
      { route: '/guide', page: 'Cyborg Guide', content: 'Guide' },
      { route: '/wiki', page: 'Wiki', content: 'Wiki' },
      { route: '/gilbertogil', page: 'Gilberto Gil', content: 'Gilberto' },
      { route: '/futurivel', page: 'Futurivel', content: 'Futurível' },
    ];

    primaryRoutes.forEach(({ route, page, content }) => {
      it(`should load ${page} page successfully`, () => {
        cy.visit(`${route}?rpc=ws://localhost:8000`);

        // Page loads without errors
        cy.url().should('include', route);

        // Expected content appears
        cy.contains(content, { timeout: 10000 }).should('be.visible');
      });
    });
  });

  describe('Explore Section Routes', () => {
    const exploreSections = [
      'bidders',
      'candidates',
      'members',
      'payouts',
      'suspended',
    ];

    exploreSections.forEach((section) => {
      it(`should load /explore/${section} with blockchain data`, () => {
        cy.visitExplore(section);

        // Verify URL
        cy.url().should('include', `/explore/${section}`);

        // Verify blockchain data loaded (not loading spinner)
        cy.get('[data-testid="loading-spinner"]').should('not.exist');
        cy.get('[data-testid="blockchain-data"]').should('be.visible');
      });
    });
  });

  describe('Proof of Ink Pages', () => {
    const poiSubsections = [
      'examples',
      'rules',
      'gallery',
      'next-head',
    ];

    poiSubsections.forEach((subsection) => {
      it(`should load /explore/poi/${subsection} correctly`, () => {
        cy.visit(`/explore/poi/${subsection}?rpc=ws://localhost:8000`);

        // Page loads without errors
        cy.url().should('include', `/explore/poi/${subsection}`);

        // Content loads
        cy.get('main').should('be.visible');
      });
    });
  });

  describe('RPC Parameter Preservation', () => {
    it('should persist RPC parameter across navigation', () => {
      // Start on bidders page with RPC parameter
      cy.visit('/explore/bidders?rpc=ws://localhost:8000');
      cy.waitForBlockchainData();

      // Navigate to Candidates via link
      cy.get('[data-testid="nav-link-candidates"]').click();

      // Verify URL includes RPC parameter
      cy.url().should('include', '/explore/candidates');
      cy.url().should('include', 'rpc=ws://localhost:8000');

      // Verify blockchain data loads from local RPC
      cy.waitForBlockchainData();
      cy.get('[data-testid="candidates-list"]').should('be.visible');
    });

    it('should maintain RPC parameter across multiple transitions', () => {
      cy.visit('/explore/bidders?rpc=ws://localhost:8000');
      cy.waitForBlockchainData();

      // Navigate through multiple pages
      const pages = ['candidates', 'members', 'payouts'];

      pages.forEach((page) => {
        cy.get(`[data-testid="nav-link-${page}"]`).click();
        cy.url().should('include', 'rpc=ws://localhost:8000');
        cy.waitForBlockchainData();
      });
    });
  });
});
```
