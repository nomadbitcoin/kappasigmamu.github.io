### Cypress Task Implementation

```typescript
// cypress.config.ts or cypress/plugins/index.js
import { defineConfig } from 'cypress';
import { exec } from 'child_process';
import { promisify } from 'util';

const execAsync = promisify(exec);

export default defineConfig({
  e2e: {
    setupNodeEvents(on, config) {
      on('task', {
        async resetChopsticks() {
          // Kill existing chopsticks process
          try {
            await execAsync('pkill -f chopsticks');
          } catch (e) {
            // Process might not be running
          }

          // Restart chopsticks with fresh state
          // This assumes chopsticks is run separately
          // Alternatively, can delete DB file to reset

          return null;
        }
      });
    }
  }
});
```

### Test Isolation Pattern

```typescript
// Example in test files
beforeEach(() => {
```
